    <script>

        const STORAGE_KEY = 'byd_simple_campaign_sheet_v2';

 

        const REGION_NAMES = {

            NATIONAL: 'National',

            VIC_TAS: 'VIC / TAS',

            NSW_ACT: 'NSW / ACT',

            QLD: 'QLD',

            WA: 'WA',

            SA_NT: 'SA / NT'

        };

 

        const REGION_DEFAULTS = {

            NATIONAL: { dealers: 20, target: 600, conv: 10 },

            VIC_TAS: { dealers: 5, target: 160, conv: 10 },

            NSW_ACT: { dealers: 6, target: 190, conv: 10 },

            QLD: { dealers: 4, target: 120, conv: 10 },

            WA: { dealers: 3, target: 80, conv: 10 },

            SA_NT: { dealers: 2, target: 50, conv: 10 }

        };

 

        const TAG_STYLES = {

            POS: 'bg-amber-50 text-amber-700 border-amber-200',

            ONLINE: 'bg-sky-50 text-sky-700 border-sky-200',

            POSTERS: 'bg-purple-50 text-purple-700 border-purple-200',

            DISPLAY: 'bg-rose-50 text-rose-700 border-rose-200'

        };

 

        function getDefaultItems() {

            return [

                { id: '1', tag: 'POS', text: 'Hero Vehicle Display Stand', checked: false },

                { id: '2', tag: 'POS', text: 'Showroom Window Decals & Counter QR Cards', checked: false },

                { id: '3', tag: 'ONLINE', text: 'Geo-fenced Meta & Google Lead Ads', checked: false },

                { id: '4', tag: 'ONLINE', text: 'Customer Email & SMS Campaign & 5-Min Lead SLA Check', checked: false },

                { id: '5', tag: 'POSTERS', text: 'Showroom Offer Posters & Hanging Tags', checked: false },

                { id: '6', tag: 'DISPLAY', text: 'Vehicle Promo Display & Wrapped Demo', checked: false }

            ];

        }

 

        function buildInitialState() {

            const regions = {};

            Object.keys(REGION_NAMES).forEach(code => {

                regions[code] = {

                    items: getDefaultItems(),

                    ...REGION_DEFAULTS[code]

                };

            });

            return { currentRegion: 'NATIONAL', regions };

        }

 

        function loadState() {

            // Check if a shared state was passed in the URL hash first

            try {

                if (window.location.hash && window.location.hash.startsWith('#share=')) {

                    const encoded = window.location.hash.slice(7);

                    const jsonStr = decodeURIComponent(escape(atob(encoded)));

                    const shared = JSON.parse(jsonStr);

                    if (shared && shared.regionCode && shared.regionData) {

                        const base = buildInitialState();

                        base.currentRegion = shared.regionCode;

                        base.regions[shared.regionCode] = shared.regionData;

                        return base;

                    }

                }

            } catch (e) {

                console.warn('Invalid share link payload', e);

            }

 

            try {

                const raw = localStorage.getItem(STORAGE_KEY);

                if (!raw) return buildInitialState();

                const parsed = JSON.parse(raw);

                if (!parsed || !parsed.regions) return buildInitialState();

                return parsed;

            } catch (e) {

                return buildInitialState();

            }

        }

 

        function toggleRulesNote() {

            const note = document.getElementById('oemRulesNote');

            if (note) note.classList.toggle('hidden');

        }

 

        function shareCurrentState() {

            try {

                const payload = {

                    regionCode: state.currentRegion,

                    regionData: getActiveRegionData()

                };

                const encoded = btoa(unescape(encodeURIComponent(JSON.stringify(payload))));

                const shareUrl = `${window.location.origin}${window.location.pathname}#share=${encoded}`;

                window.history.replaceState(null, '', `#share=${encoded}`);

 

                // Copy via fallback-safe textarea for iframe compatibility

                const tempInput = document.createElement('textarea');

                tempInput.value = shareUrl;

                tempInput.style.position = 'fixed';

                tempInput.style.opacity = '0';

                document.body.appendChild(tempInput);

                tempInput.select();

                document.execCommand('copy');

                document.body.removeChild(tempInput);

 

                showToast(`Shareable ${REGION_NAMES[state.currentRegion]} link copied!`);

            } catch (e) {

                showToast('Could not generate share link');

            }

        }

 

        let state = loadState();

 

        function saveState() {

            try {

                localStorage.setItem(STORAGE_KEY, JSON.stringify(state));

            } catch (e) {

                console.warn('Storage unavailable', e);

            }

        }

 

        function showToast(msg) {

            const toast = document.getElementById('toast');

            if (!toast) return;

            toast.textContent = msg;

            toast.classList.remove('opacity-0');

            clearTimeout(window._toastTimer);

            window._toastTimer = setTimeout(() => toast.classList.add('opacity-0'), 2000);

        }

 

        function escapeHtml(str) {

            return String(str)

                .replace(/&/g, '&amp;')

                .replace(/</g, '&lt;')

                .replace(/>/g, '&gt;')

                .replace(/"/g, '&quot;');

        }

 

        function getActiveRegionData() {

            return state.regions[state.currentRegion] || state.regions.NATIONAL;

        }

 

        function renderChecklist() {

            const region = getActiveRegionData();

            const listEl = document.getElementById('checklistItems');

            const progressEl = document.getElementById('checklistProgress');

            if (!listEl) return;

 

            const doneCount = region.items.filter(i => i.checked).length;

            progressEl.textContent = `${doneCount} / ${region.items.length} Done`;

 

            if (region.items.length === 0) {

                listEl.innerHTML = `<li class="py-6 text-center text-sm text-slate-400">No assets in this checklist. Add one below.</li>`;

                return;

            }

 

            listEl.innerHTML = region.items.map(item => {

                const badgeClass = TAG_STYLES[item.tag] || 'bg-slate-100 text-slate-700 border-slate-200';

                return `

                    <li class="py-3 flex items-center gap-3 group">

                        <input type="checkbox" ${item.checked ? 'checked' : ''}

                               onchange="toggleItem('${item.id}')"

                               class="w-5 h-5 rounded border-slate-300 text-teal-600 focus:ring-teal-500 cursor-pointer shrink-0">

                        <span class="text-[11px] font-bold px-2 py-0.5 rounded border ${badgeClass} shrink-0 w-16 text-center">

                            ${escapeHtml(item.tag)}

                        </span>

                        <span contenteditable="true"

                              onblur="updateItemText('${item.id}', this.textContent)"

                              onkeydown="if(event.key==='Enter'){event.preventDefault();this.blur();}"

                              class="flex-1 text-sm font-medium outline-none rounded px-1.5 py-0.5 focus:bg-slate-100 focus:ring-1 focus:ring-teal-500 ${item.checked ? 'line-through text-slate-400' : 'text-slate-800'}">

                            ${escapeHtml(item.text)}

                        </span>

                        <button type="button" onclick="deleteItem('${item.id}')" aria-label="Remove asset"

                                class="text-slate-300 hover:text-rose-600 font-bold text-lg leading-none px-1.5 transition no-print">

                            &times;

                        </button>

                    </li>

                `;

            }).join('');

        }

 

        function toggleItem(id) {

            const region = getActiveRegionData();

            const item = region.items.find(i => i.id === id);

            if (item) {

                item.checked = !item.checked;

                saveState();

                renderChecklist();

            }

        }

 

        function updateItemText(id, newText) {

            const region = getActiveRegionData();

            const item = region.items.find(i => i.id === id);

            const cleaned = newText.trim();

            if (item && cleaned && item.text !== cleaned) {

                item.text = cleaned;

                saveState();

                showToast('Asset updated');

            } else {

                renderChecklist();

            }

        }

 

        function deleteItem(id) {

            const region = getActiveRegionData();

            region.items = region.items.filter(i => i.id !== id);

            saveState();

            renderChecklist();

            showToast('Asset removed');

        }

 

        function calculateAndRender() {

            const region = getActiveRegionData();

            const dealers = Math.max(1, parseInt(region.dealers, 10) || 1);

            const target = Math.max(1, parseInt(region.target, 10) || 1);

            const conv = Math.min(100, Math.max(0.5, parseFloat(region.conv) || 10));

 

            const leads = Math.ceil(target / (conv / 100));

            const leadsPerDealer = Math.ceil(leads / dealers);

            const drives = Math.ceil(target / 0.30);

            const stock = Math.ceil(target * 1.35);

            const stockPerDealer = Math.ceil(stock / dealers);

 

            const posters = dealers * 3;

            const flags = dealers * 4;

            const brochures = Math.ceil(drives * 1.5);

            const displays = dealers * 1;

            const minDemos = dealers * 2;

            const maxDemos = dealers * 3;

 

            document.getElementById('outLeads').textContent = leads.toLocaleString();

            document.getElementById('outLeadsSub').textContent = `${leadsPerDealer.toLocaleString()} leads / dealership`;

            document.getElementById('outDrives').textContent = drives.toLocaleString();

            document.getElementById('outStock').textContent = stock.toLocaleString();

            document.getElementById('outStockSub').textContent = `1.35x target cover (${stockPerDealer.toLocaleString()} / dealer)`;

 

            document.getElementById('matPosters').textContent = posters.toLocaleString();

            document.getElementById('matFlags').textContent = flags.toLocaleString();

            document.getElementById('matBrochures').textContent = brochures.toLocaleString();

            document.getElementById('matDisplays').textContent = displays.toLocaleString();

            document.getElementById('matDemos').textContent = `${minDemos.toLocaleString()} – ${maxDemos.toLocaleString()}`;

        }

 

        function renderAll() {

            const region = getActiveRegionData();

            document.getElementById('regionSelect').value = state.currentRegion;

            document.getElementById('activeRegionPill').textContent = REGION_NAMES[state.currentRegion] || state.currentRegion;

            document.getElementById('inputDealers').value = region.dealers;

            document.getElementById('inputTarget').value = region.target;

            document.getElementById('inputConv').value = region.conv;

            renderChecklist();

            calculateAndRender();

        }

 

        function openResetModal() {

            const modal = document.getElementById('resetModal');

            modal.classList.remove('hidden');

            modal.classList.add('flex');

        }

 

        function closeResetModal() {

            const modal = document.getElementById('resetModal');

            modal.classList.add('hidden');

            modal.classList.remove('flex');

        }

 

        function confirmResetRegion() {

            const code = state.currentRegion;

            state.regions[code] = {

                items: getDefaultItems(),

                ...REGION_DEFAULTS[code]

            };

            saveState();

            renderAll();

            closeResetModal();

            showToast(`${REGION_NAMES[code]} reset to defaults`);

        }

 

        document.getElementById('regionSelect').addEventListener('change', (e) => {

            state.currentRegion = e.target.value;

            saveState();

            renderAll();

        });

 

        document.getElementById('addAssetForm').addEventListener('submit', (e) => {

            e.preventDefault();

            const tagEl = document.getElementById('newAssetTag');

            const textEl = document.getElementById('newAssetText');

            const text = textEl.value.trim();

            if (!text) return;

 

            const region = getActiveRegionData();

            region.items.push({

                id: String(Date.now()),

                tag: tagEl.value,

                text,

                checked: false

            });

            textEl.value = '';

            saveState();

            renderChecklist();

            showToast('Asset added');

        });

 

        ['inputDealers', 'inputTarget', 'inputConv'].forEach(inputId => {

            document.getElementById(inputId).addEventListener('input', (e) => {

                const region = getActiveRegionData();

                const val = parseFloat(e.target.value);

                if (inputId === 'inputDealers') region.dealers = isNaN(val) ? 1 : val;

                if (inputId === 'inputTarget') region.target = isNaN(val) ? 1 : val;

                if (inputId === 'inputConv') region.conv = isNaN(val) ? 1 : val;

                saveState();

                calculateAndRender();

            });

        });

 

        renderAll();
